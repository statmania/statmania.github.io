    \ifthenelse{\equal{#5}{a}}
    {\begin{answer}\thequestion. (a)~#1\end{answer}}

        {\begin{answer}\thequestion. (b)~#2\end{answer}}

\question \textbf{What is the probability of a certain event (an event guaranteed to occur)?}
\choice{0}{0.5}{1}{Undefined}{c}

\question \textbf{If two events $A$ and $B$ are mutually exclusive (disjoint), which of the following is true?}
\choice{$P(A \cap B) = P(A)P(B)$}{$P(A \cap B) = 0$}{$P(A \cup B) = 1$}{$P(A|B) = P(A)$}{b}

\question \textbf{Which of the following statements regarding probability are correct?}

i. $0 \le P(A) \le 1$ for any event $A$ \\
ii. $P(S) = 1$ where $S$ is the sample space \\
iii. For mutually exclusive events $A$ and $B$, $P(A \cup B) = P(A) + P(B)$

\textbf{Which one is correct?}

\choice{i and ii}{i and iii}{ii and iii}{i, ii and iii}{d}

\textbf{Answer the next TWO questions based on the following information}

In a group of 100 students, 60 passed in Mathematics, 50 passed in Statistics, and 30 passed in both subjects. A student is selected at random.

\question \textbf{What is the probability that the student passed in Mathematics or Statistics?}
\choice{0.80}{0.50}{0.30}{0.70}{a}

\question \textbf{What is the conditional probability that the student passed in Statistics given that s/he passed in Mathematics?}
\choice{0.30}{0.50}{0.60}{0.80}{b}

\question \textbf{Which of the following is an example of a continuous random variable?}
\choice{Number of defective items in a batch}{Height of students in a class}{Number of phone calls received in an hour}{Outcome of rolling a die}{b}

\question \textbf{What is $F(\infty)$ for any valid cumulative distribution function (CDF) $F(x)$?}
\choice{$-\infty$}{-1}{0}{1}{d}

\question \textbf{Which of the following are necessary conditions for a discrete probability function $P(x)$?}

i. $P(x_i) \ge 0$ for all $x_i$ \\
ii. $\sum P(x_i) = 1$ \\
iii. $P(x_i) \le 1$ for all $x_i$

\textbf{Which one is correct?}

\choice{i and ii}{i and iii}{ii and iii}{i, ii and iii}{d}

\textbf{Answer the next TWO questions based on the following information}

A discrete random variable $X$ has the following probability distribution:
\begin{table}[H]
\centering
\begin{tabular}{c|c|c|c}
$X$ & 0 & 1 & 2 \\ \hline
$P(x)$ & 0.50 & 0.25 & 0.25
\end{tabular}
\end{table}

\question \textbf{What is the value of the cumulative distribution function $F(1)$?}
\choice{0.25}{0.50}{0.75}{1.00}{c}

\question \textbf{What is $P(X \ge 1)$?}
\choice{0.25}{0.50}{0.75}{1.00}{c}

\question \textbf{If $X$ is a random variable and $c$ is a constant, what is the variance $V(c)$?}
\choice{$c$}{$c^2$}{0}{1}{c}

\question \textbf{If $E(X) = 4$ and $V(X) = 5$, what is the value of $E(X^2)$?}
\choice{9}{16}{21}{25}{c}

\question \textbf{If $Y = 3X + 5$ and $V(X) = 4$, what is the variance $V(Y)$?}
\choice{17}{36}{41}{12}{b}

\question \textbf{For two independent random variables $X$ and $Y$, which of the following properties are true?}

i. $E(X + Y) = E(X) + E(Y)$ \\
ii. $E(XY) = E(X)E(Y)$ \\
iii. $V(X - Y) = V(X) + V(Y)$

\textbf{Which one is correct?}

\choice{i and ii}{i and iii}{ii and iii}{i, ii and iii}{d}

\question \textbf{If $E(X) = 5$, what is the value of $E(2X - 3)$?}
\choice{7}{10}{13}{5}{a}

\question \textbf{The first central moment ($\mu_1$) of any frequency distribution is always equal to --}
\choice{1}{0}{-1}{Mean}{b}

\question \textbf{Which moment is used to measure the variance (dispersion) of a distribution?}
\choice{First raw moment ($\mu_1'$)}{First central moment ($\mu_1$)}{Second central moment ($\mu_2$)}{Third central moment ($\mu_3$)}{c}

\question \textbf{For a mesokurtic (normal-shaped) distribution, what is the value of kurtosis $\beta_2$?}
\choice{0}{1}{3}{-3}{c}

\question \textbf{Which of the following statements about moments are correct?}

i. The first raw moment around zero ($\mu_1'$) is the arithmetic mean. \\
ii. The third central moment ($\mu_3$) can be negative, zero, or positive. \\
iii. The second central moment ($\mu_2$) can never be negative.

\textbf{Which one is correct?}

\choice{i and ii}{i and iii}{ii and iii}{i, ii and iii}{d}

\question \textbf{If the first raw moment about 4 is $-2$, what is the value of the arithmetic mean?}
\choice{2}{-2}{6}{4}{a}

\question \textbf{The correlation coefficient $r$ between two variables lies within which range?}
\choice{$0 \le r \le 1$}{$-1 \le r \le 1$}{$-\infty < r < \infty$}{$-1 \le r \le 0$}{b}

\question \textbf{If the correlation coefficient between $X$ and $Y$ is $r = 0.6$, what is the correlation coefficient between $2X + 3$ and $3Y - 5$?}
\choice{0.6}{1.2}{0.2}{Cannot be determined}{a}

\question \textbf{The correlation coefficient $r$ is the geometric mean of --}
\choice{Two means}{Two variances}{Two regression coefficients ($b_{yx}$ and $b_{xy}$)}{Two standard deviations}{c}

\question \textbf{Which of the following statements are true about the correlation coefficient $r$?}

i. It is independent of change of origin and scale. \\
ii. It is a unitless measure. \\
iii. Correlation implies a causal relationship between variables.

\textbf{Which one is correct?}

\choice{i and ii}{i and iii}{ii and iii}{i, ii and iii}{a}

\choice{0.32}{0.60}{0.70}{0.80}{b}
